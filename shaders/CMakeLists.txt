set(LB_SHADER_OUT_DIR "${CMAKE_BINARY_DIR}/shaders")
set(LB_GEN_DIR "${CMAKE_BINARY_DIR}/gen")
file(MAKE_DIRECTORY "${LB_SHADER_OUT_DIR}" "${LB_GEN_DIR}")

file(GLOB LB_SHARED_SLANG CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/shared/*.slang")
file(GLOB LB_PASS_SLANG   CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.slang")
list(FILTER LB_PASS_SLANG EXCLUDE REGEX "layout_probe\\.slang$")

lb_compile_shaders(lb_shaders
  OUTPUT_DIR "${LB_SHADER_OUT_DIR}"
  SOURCES ${LB_PASS_SLANG}
  SHARED ${LB_SHARED_SLANG}
  INCLUDE_DIRS "${CMAKE_CURRENT_SOURCE_DIR}/shared" "${CMAKE_CURRENT_SOURCE_DIR}")

lb_generate_layouts(lb_layouts_gen
  PROBE "${CMAKE_CURRENT_SOURCE_DIR}/layout_probe.slang"
  OUTPUT "${LB_GEN_DIR}/layouts.gen.h"
  SHARED ${LB_SHARED_SLANG}
  INCLUDE_DIRS "${CMAKE_CURRENT_SOURCE_DIR}/shared" "${CMAKE_CURRENT_SOURCE_DIR}")

set(LB_GEN_DIR "${LB_GEN_DIR}" PARENT_SCOPE)
set(LB_SHADER_OUT_DIR "${LB_SHADER_OUT_DIR}" PARENT_SCOPE)
